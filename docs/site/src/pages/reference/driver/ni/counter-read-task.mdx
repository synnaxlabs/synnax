## Important Rules

- **One task per module** -> Only one running task can claim a module at a time.
- **Unique ports** -> A counter port can appear only once in a task.
- **Stream rate** -> For sample rates under 50 Hz, set the stream rate equal to the
  sample rate. Above that, keep the stream rate under 50 Hz.

```python
import synnax as sy
from synnax import ni

client = sy.Synnax()

# Retrieve device
counter_dev = client.devices.retrieve(name="Mod1_Counter")

# Create index channel
ctr_time = client.channels.create(
    name="ctr_time",
    is_index=True,
    data_type=sy.DataType.TIMESTAMP,
    retrieve_if_name_exists=True,
)

# Create data channels
frequency_ch = client.channels.create(
    name="frequency",
    index=ctr_time.key,
    data_type=sy.DataType.FLOAT32,
    retrieve_if_name_exists=True,
)
edge_count_ch = client.channels.create(
    name="edge_count",
    index=ctr_time.key,
    data_type=sy.DataType.UINT32,
    retrieve_if_name_exists=True,
)
angular_pos_ch = client.channels.create(
    name="angular_position",
    index=ctr_time.key,
    data_type=sy.DataType.FLOAT32,
    retrieve_if_name_exists=True,
)

# Create and configure task
task = ni.CounterReadTask(
    name="NI Counter Read Task",
    sample_rate=sy.Rate.HZ * 100,
    stream_rate=sy.Rate.HZ * 25,
    channels=[
        ni.CIFrequencyChannel(
            channel=frequency_ch.key,
            device=counter_dev.key,
            port=0,
            min_val=1,
            max_val=10000,
            units="Hz",
            edge="Rising",
            meas_method="DynamicAvg",
        ),
        ni.CIEdgeCountChannel(
            channel=edge_count_ch.key,
            device=counter_dev.key,
            port=1,
            active_edge="Rising",
            count_direction="CountUp",
            initial_count=0,
        ),
        ni.CIPositionAngularChannel(
            channel=angular_pos_ch.key,
            device=counter_dev.key,
            port=2,
            units="Degrees",
            decoding_type="X4",
            pulses_per_rev=1024,
            initial_angle=0.0,
            z_index_enabled=False,
        ),
    ],
)

client.tasks.configure(task)

# Start task and read data
with task.run():
    with client.open_streamer(["frequency", "edge_count", "angular_position"]) as streamer:
        for _ in range(10):
            frame = streamer.read()
            print(frame)
```

```typescript
import { Synnax } from "@synnaxlabs/client";

const client = new Synnax();

// Retrieve device
const [counterDev] = await client.devices.retrieve({ names: ["Mod1_Counter"] });

// Create index channel
const ctrTime = await client.channels.create(
  {
    name: "ctr_time",
    isIndex: true,
    dataType: "timestamp",
  },
  { retrieveIfNameExists: true },
);

// Create data channels
const frequencyCh = await client.channels.create(
  {
    name: "frequency",
    index: ctrTime.key,
    dataType: "float32",
  },
  { retrieveIfNameExists: true },
);

const edgeCountCh = await client.channels.create(
  {
    name: "edge_count",
    index: ctrTime.key,
    dataType: "uint32",
  },
  { retrieveIfNameExists: true },
);

const angularPosCh = await client.channels.create(
  {
    name: "angular_position",
    index: ctrTime.key,
    dataType: "float32",
  },
  { retrieveIfNameExists: true },
);

// Create and configure task
const task = await client.tasks.create({
  name: "NI Counter Read Task",
  rack: counterDev.rack,
  type: "ni_counter_read",
  config: {
    sampleRate: 100,
    streamRate: 25,
    channels: [
      {
        type: "ci_frequency",
        channel: frequencyCh.key,
        device: counterDev.key,
        port: 0,
        minVal: 1,
        maxVal: 10000,
        units: "Hz",
        edge: "Rising",
        measMethod: "DynamicAvg",
      },
      {
        type: "ci_edge_count",
        channel: edgeCountCh.key,
        device: counterDev.key,
        port: 1,
        activeEdge: "Rising",
        countDirection: "CountUp",
        initialCount: 0,
      },
      {
        type: "ci_position_angular",
        channel: angularPosCh.key,
        device: counterDev.key,
        port: 2,
        units: "Degrees",
        decodingType: "X4",
        pulsesPerRev: 1024,
        initialAngle: 0.0,
        zIndexEnabled: false,
      },
    ],
  },
});

// Start task
await task.start();

// Read data
const streamer = await client.openStreamer([
  "frequency",
  "edge_count",
  "angular_position",
]);
for (let i = 0; i < 10; i++) {
  const frame = await streamer.read();
  console.log(frame);
}

// Stop task
await task.stop();
await streamer.close();
```
